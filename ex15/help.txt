# Основные логические формулы:

# A * 0 = 0
# A * 1 = A     A - логическая константа
# A + 0 = A
# A + 1 = 1
# A * A = A
# A + A = 0
# not(not(A)) = A
# A * not(A) = 0
# A + not(A) = 1
# A * N = B * A
# A + B = B + A

# Сочитательный закон(ассоциативность)
# (A * B) * C = A * (B * C) = A * B * C
# (A + B) + C = A + (B + C) = A + B + C

# Распределительный закон(дистрибутивность)
# A * (B + C) = A * B + A * C
# A + B * C = (A + B) * (A + C)
# (A + B) * (C + D) = A * C + A * D + B * C + B * D

# Законы де Моргана
# not(A * B) = not(a) + not(B)
# not(A + B) = not(a) * not(B)

#----------------------
# Замена импликации   |
# A <= B = not(A) + B |
#----------------------

# Замена эквиваленции
# A == B = (A <= B) * (B <= A)
# A == B = A * B + not(A) * not(B)
# not(A == B) = A * not(B) + not(A) * B
